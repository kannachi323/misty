88 inline std::string strip_trailing_separators(std::string path) {
89 while (path.size() > 1 && (path.back() == '/' || path.back() == '\\')) {
90 if (path.size() == 3 && path[1] == ':') {
91 break;
92 }
93 path.pop_back();
94 }
95 return path;
96 }

101 inline std::string normalize_for_history(const std::string& path) {
102 if (path.empty() || path.rfind("misty://", 0) == 0) {
103 return strip_trailing_separators(path);
104 }
105
106 std::error_code ec;
107 fs::path normalized = fs::weakly_canonical(fs::path(path), ec);
108 if (ec) {
109 normalized = fs::path(path).lexically_normal();
110 }
111 return strip_trailing_separators(normalized.generic_string());
112 }
113
117 inline bool same_history_path(const std::string& lhs, const std::string& rhs) {
118 return normalize_for_history(lhs) == normalize_for_history(rhs);
119 }

124 inline std::vector<std::string> split_path(const std::string& path) {
125 std::vector<std::string> components;
126 std::string current;
127 for (char c : path) {
128 if (c == '/') {
129 if (!current.empty()) {
130 components.push_back(current);
131 current.clear();
132 }
133 } else {
134 current += c;
135 }
136 }
137 if (!current.empty()) {
138 components.push_back(current);
139 }
140 return components;
141 }

366 inline void navigate_to_local_path(FileExplorerState& state, const std::string& path, bool update_history = true) {
367 state.is_loading = true;
368 state.cancel_loading_animation_cycle();
369
370 try {
371 std::string new_path = path_utf8_generic_string(fs::canonical(fs::path(path)));
372 if (new_path == state.current_path) {
373 update_history = false;
374 }
375
376 std::vector<FileItem> new_files;
377 for (const auto& entry : fs::directory_iterator(path)) {
378 std::string fname = path_utf8_filename(entry.path());
379 if (!state.show_hidden && !fname.empty() && fname[0] == '.') continue;
380
381 FileItem item;
382 item.path = path_utf8_generic_string(entry.path());
383 item.id = item.path;
384 item.name = fname;
385 item.is_dir = entry.is_directory();
386 item.status = SyncStatus::LOCAL;
387
388 if (!item.is_dir) {
389 try {
390 item.size = fs::file_size(entry.path());
391 } catch (...) {
392 item.size = 0;
393 }
394 }
395
396 try {
397 auto ftime = fs::last_write_time(entry.path());
398 auto sctp = std::chrono::time_point_cast<std::chrono::system_clock::duration>(
399 ftime - fs::file_time_type::clock::now() + std::chrono::system_clock::now()
400 );
401 auto time_t_val = std::chrono::system_clock::to_time_t(sctp);
402 char buf[32];
403 std::strftime(buf, sizeof(buf), "%Y-%m-%d %H:%M", std::localtime(&time_t_val));
404 item.last_modified = buf;
405 } catch (...) {
406 item.last_modified = "";
407 }
408
409 new_files.push_back(item);
410 }
411
412 if (update_history) {
413 std::string current_path_str(state.current_path);
414 if (!current_path_str.empty() && current_path_str != new_path) {
415 state.back_history.push(current_path_str);
416 }
417 while (!state.forward_history.empty()) {
418 state.forward_history.pop();
419 }
420 }
421
422 state.files = std::move(new_files);
423 strncpy(state.current_path, new_path.c_str(), sizeof(state.current_path) - 1);
424 state.current_path[sizeof(state.current_path) - 1] = '\0';
425 strncpy(state.search_path, new_path.c_str(), sizeof(state.search_path) - 1);
426
427 state.is_loading = false;
428 state.cancel_loading_animation_cycle();
429 state.sort_dirty = true;
430 state.selected_files.clear();
431 state.last_selected_index = -1;
432 state.error_msg = "";
433 }
434 catch (const std::exception& e) {
435 state.error_msg = e.what();
436 state.is_loading = false;
437 state.cancel_loading_animation_cycle();
438 state.sort_dirty = true;
439 }
440 }
