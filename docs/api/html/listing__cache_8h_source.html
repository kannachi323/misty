9// listing_cache: simple on-disk cache for remote folder listings, used for
10// stale-while-revalidate rendering in the file explorer.
11//
12// Layout: ~/.misty/.cache/remotes/<remote_name>/<encoded_path>.json
13//
14// The remote_path is URL-style encoded so '/' becomes %2F, keeping every
15// listing in a flat per-remote directory and avoiding file/directory name
16// collisions (e.g. caching both "Documents" and "Documents/Work").
17
18namespace misty::core::listing_cache {
19
20 namespace fs = std::filesystem;
21
22 inline std::string root_dir() {
23 const char* home = std::getenv("HOME");
24 if (!home) home = ".";
25 return std::string(home) + "/.misty/.cache/remotes";
26 }
27
28 inline std::string legacy_root_dir() {
29 const char* home = std::getenv("HOME");
30 if (!home) home = ".";
31 return std::string(home) + "/.misty/.cache/listings";
32 }
33
34 // URL-style encode a path so it survives as a single filename component.
35 inline std::string encode_path(const std::string& path) {
36 if (path.empty()) return "__root__";
37 std::ostringstream out;
38 out << std::hex << std::uppercase;
39 for (unsigned char c : path) {
40 bool safe = (c >= 'a' && c <= 'z') ||
41 (c >= 'A' && c <= 'Z') ||
42 (c >= '0' && c <= '9') ||
43 c == '-' || c == '.' || c == '_' || c == '~';
44 if (safe) {
45 out << static_cast<char>(c);
46 } else {
47 out << '%';
48 if (c < 0x10) out << '0';
49 out << static_cast<int>(c);
50 }
51 }
52 return out.str();
53 }
54
55 inline std::string remote_dir(const std::string& remote) {
56 return root_dir() + "/" + remote;
57 }
58
59 inline std::string file_for(const std::string& remote, const std::string& path) {
60 return remote_dir(remote) + "/" + encode_path(path) + ".json";
61 }
62
63 inline std::string legacy_file_for(const std::string& remote, const std::string& path) {
64 return legacy_root_dir() + "/" + remote + "/" + encode_path(path) + ".json";
65 }

67 // Returns true on hit; out_body receives the cached JSON body verbatim.
68 inline bool load(const std::string& remote, const std::string& path, std::string& out_body) {
69 if (remote.empty()) return false;
70 std::ifstream f(file_for(remote, path), std::ios::binary);
71 if (!f) {
72 f.open(legacy_file_for(remote, path), std::ios::binary);
73 }
74 if (!f) return false;
75 std::ostringstream ss;
76 ss << f.rdbuf();
77 out_body = ss.str();
78 return !out_body.empty();
79 }
80
81 inline void save(const std::string& remote, const std::string& path, const std::string& body) {
82 if (remote.empty() || body.empty()) return;
83 std::error_code ec;
84 fs::create_directories(remote_dir(remote), ec);
85 std::ofstream f(file_for(remote, path), std::ios::binary | std::ios::trunc);
86 if (f) f.write(body.data(), static_cast<std::streamsize>(body.size()));
87 }
88
89 // Best-effort: nuke a remote's entire cache (e.g. when the remote is
90 // disconnected). Failures are silent.
91 inline void clear_remote(const std::string& remote) {
92 if (remote.empty()) return;
93 std::error_code ec;
94 fs::remove_all(remote_dir(remote), ec);
95 }
96
97 inline void clear(const std::string& remote, const std::string& path) {
98 if (remote.empty()) return;
99 std::error_code ec;
100 fs::remove(file_for(remote, path), ec);
101 }
